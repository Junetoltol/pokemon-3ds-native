cmake_minimum_required(VERSION 3.24)
project(pokemon_3ds_native VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(SDL3 CONFIG REQUIRED)
add_executable(pokemon-3ds-native MACOSX_BUNDLE src/main.cpp)
target_link_libraries(pokemon-3ds-native PRIVATE SDL3::SDL3)
set_target_properties(pokemon-3ds-native PROPERTIES MACOSX_BUNDLE_GUI_IDENTIFIER org.pokemon3dsnative.launcher MACOSX_BUNDLE_BUNDLE_NAME "Pokemon 3DS Native")
include(CTest)
add_executable(runtime-tests tests/runtime.cpp)
target_include_directories(runtime-tests PRIVATE src)
add_test(NAME runtime-contract COMMAND runtime-tests)
install(TARGETS pokemon-3ds-native BUNDLE DESTINATION . RUNTIME DESTINATION bin)
if(WIN32)
  add_custom_command(TARGET pokemon-3ds-native POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_RUNTIME_DLLS:pokemon-3ds-native> $<TARGET_FILE_DIR:pokemon-3ds-native>
    COMMAND_EXPAND_LISTS)
endif()
find_package(Python3 COMPONENTS Interpreter)
if(Python3_FOUND)
  add_test(NAME local-tools COMMAND ${Python3_EXECUTABLE} -m unittest discover -s ${CMAKE_SOURCE_DIR}/tests -p test_*.py -v)
endif()
